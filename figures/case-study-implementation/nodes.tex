\node[stagecard,anchor=north west,text width=3.80cm]
  (stage1) at (0,0) {\begin{stagetable}
  \tabletitle{Stage 1: Binding}
  \tablecolumns
  \tablerule
  \tablesection{Input}
  \tablerow{\mathrm{SRC}_c}{Containerised reproducibility claim.}
  \tablerow{\mathrm{SRC}_n}{Uncontainerised reproducibility claim.}
  \tablerow{\mathcal I_c,\mathcal I_n}{General claims over three contexts each.}
  \tablerow{\mathrm{inputs}}{Five external sources, identified and constrained
    by size, format, and digest.}
  \tablerule
  \tablesection{Operation}
  \tablerow{\bindop}{Resolve both claims and their inference requirements
    before attempts begin.}
  \tablerule
  \tablesection{Output}
  \tablerow{\kappa}{Complete: shared by all attempts; incomplete: returned as
    an input-level result and assessment stops.}
  \end{stagetable}};

\node[stagecard,anchor=north west,text width=3.80cm]
  (stage2) at ([xshift=0.28cm]stage1.north east) {\begin{stagetable}
  \tabletitle{Stage 2: Registration}
  \tablecolumns
  \tablerule
  \tablesection{Input}
  \tablerow{r}{Six attempts: two modes $\times$ three runners.}
  \tablerow{R_c}{One digest-qualified image on Ubuntu 22/x64,
    Ubuntu 24/x64, and Ubuntu 24/ARM64 via QEMU.}
  \tablerow{R_n}{Environment rebuilt on Ubuntu 24/x64,
    Windows 2025/x64, and macOS 15/ARM64.}
  \tablerow{\mathrm{selection}_r}{All environments are selected in advance.}
  \tablerule
  \tablesection{Operation}
  \tablerow{\registerop}{Preregister each runner attempt before execution.}
  \tablerule
  \tablesection{Output}
  \tablerow{\rho_r}{Mode, realised runner context, source, selection basis,
    and conformance to $R$.}
  \end{stagetable}};

\node[stagecard,anchor=north west,text width=3.80cm]
  (stage3) at ([xshift=0.28cm]stage2.north east) {\begin{stagetable}
  \tabletitle{Stage 3: Execute and assess}
  \tablecolumns
  \tablerule
  \tablesection{Input}
  \tablerow{B_c}{Fixed image, analysis, figure, and \LaTeX.}
  \tablerow{B_n}{Reconstructed Python environment, analysis, and figure;
    image and \LaTeX{} excluded.}
  \tablerow{\mathcal P}{Runner, environment, inputs, execution, and output
    lineage must be recorded.}
  \tablerow{\mathcal A}{Analysis and figure data exact; rendered figure
    checked for format and dimensions.}
  \tablerow{\delta}{Same-index MNIST--QMNIST prediction agreement
    $\geq 99\%$.}
  \tablerule
  \tablesection{Operation}
  \tablerow{\verifyop}{Execute $W$; realise scope and context; check
    conformance and evidence; decide.}
  \tablerule
  \tablesection{Output}
  \tablerow{x_r}{Predictions, metrics, figure, and container-only PDF.}
  \tablerow{P_r,a_r}{Recorded provenance and attributable evidence.}
  \tablerow{y_r}{Attempt classification, reproduction decision, and
    justification.}
  \end{stagetable}};

\node[stagecard,anchor=north west,text width=3.80cm]
  (stage4) at ([xshift=0.28cm]stage3.north east) {\begin{stagetable}
  \tabletitle{Stage 4: Infer and release}
  \tablecolumns
  \tablerule
  \tablesection{Input}
  \tablerow{\mathcal L}{All six registered attempt assessments.}
  \tablerow{M}{Retain missing, inadmissible, and unassessable attempts.}
  \tablerow{g}{Support requires every attempt to reproduce and all exact
    outputs to agree.}
  \tablerule
  \tablesection{Operation}
  \tablerow{}{Apply
    ${\collectop}\rightarrow{\inferop}\rightarrow{\reportop}$ separately to
    $\mathcal I_c$ and $\mathcal I_n$.}
  \tablerule
  \tablesection{Output}
  \tablerow{\gamma_c,\gamma_n}{Separate containerised and uncontainerised
    general assessments.}
  \tablerow{\mathrm{report}}{Both claims linked to all attempts, provenance,
    evidence, and justifications.}
  \tablerow{\mathrm{release}}{Publish only when
    $\gamma_c=\gamma_n=\mathrm{supported}$.}
  \end{stagetable}};
